cmake_minimum_required(VERSION 3.20)
project(occ_agent_harness VERSION 0.2.0 LANGUAGES CXX)
set(OCCT_ROOT "C:/Programs/OpenCASCADE-7.7.0-vc14-64/opencascade-7.7.0" CACHE PATH "OCCT SDK root")
get_filename_component(OCCT_PRODUCTS "${OCCT_ROOT}" DIRECTORY)
set(TCL_ROOT "${OCCT_PRODUCTS}/tcltk-86-64" CACHE PATH "Tcl 8.6 SDK")
set(RAPIDJSON_ROOT "${OCCT_PRODUCTS}/rapidjson-1.1.0" CACHE PATH "RapidJSON headers")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This build targets the supplied Windows x64 MSVC OCCT 7.7 SDK")
endif()
add_library(occ_agent MODULE native/occ_agent.cpp)
target_compile_features(occ_agent PRIVATE cxx_std_17)
target_compile_definitions(occ_agent PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(occ_agent PRIVATE /W4 /utf-8 /EHsc)
target_include_directories(occ_agent SYSTEM PRIVATE "${OCCT_ROOT}/inc" "${TCL_ROOT}/include" "${RAPIDJSON_ROOT}/include")
target_link_directories(occ_agent PRIVATE "${OCCT_ROOT}/win64/vc14/lib" "${TCL_ROOT}/lib")
target_link_libraries(occ_agent PRIVATE TKDraw TKTopTest TKTopAlgo TKBRep TKG3d TKG2d TKGeomBase TKMath TKernel TKMesh TKSTL TKSTEP TKXSBase tcl86)
set_target_properties(occ_agent PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
